```cpp
    cv::Mat generateTestImage(cv::Size size, int type) {
        cv::Mat image(size, type);
        
        if (type == CV_8UC1) {
            // 그레이스케일 패턴 생성
            for (int y = 0; y < size.height; y++) {
                uchar* ptr = image.ptr<uchar>(y);
                for (int x = 0; x < size.width; x++) {
                    ptr[x] = (x + y) % 256;
                }
            }
        } else if (type == CV_8UC3) {
            // 컬러 패턴 생성
            for (int y = 0; y < size.height; y++) {
                cv::Vec3b* ptr = image.ptr<cv::Vec3b>(y);
                for (int x = 0; x < size.width; x++) {
                    ptr[x][0] = (x * 2) % 256;      // B
                    ptr[x][1] = (y * 2) % 256;      // G
                    ptr[x][2] = (x + y) % 256;      // R
                }
            }
        }
        
        return image;
    }

                       const vector<double>& opencv_times,
                       const vector<double>& custom_times) {
        
        cout << "\n=== " << algorithm_name << " 스케일링 분석 ===" << endl;
        
        // 선형성 분석 (O(n) 복잡도 확인)
        for (int i = 1; i < pixels.size(); i++) {
            double pixel_ratio = static_cast<double>(pixels[i]) / pixels[i-1];
            double opencv_ratio = opencv_times[i] / opencv_times[i-1];
            double custom_ratio = custom_times[i] / custom_times[i-1];
            
            cout << "픽셀 증가율: " << pixel_ratio << "x" << endl;
            cout << "OpenCV 시간 증가율: " << opencv_ratio << "x" << endl;
            cout << "Custom 시간 증가율: " << custom_ratio << "x" << endl;
            cout << "OpenCV 효율성: " << (pixel_ratio / opencv_ratio) << endl;
            cout << "Custom 효율성: " << (pixel_ratio / custom_ratio) << endl;
            cout << "---" << endl;
        }
        
        // 처리량 계산 (픽셀/초)
        cout << "\n처리량 분석 (MP/s - 메가픽셀/초):" << endl;
        for (int i = 0; i < pixels.size(); i++) {
            double opencv_throughput = (pixels[i] / 1000000.0) / (opencv_times[i] / 1000.0);
            double custom_throughput = (pixels[i] / 1000000.0) / (custom_times[i] / 1000.0);
            
            cout << "크기 " << static_cast<int>(sqrt(pixels[i])) << "x" << static_cast<int>(sqrt(pixels[i])) 
                 << ": OpenCV=" << opencv_throughput << " MP/s, Custom=" << custom_throughput << " MP/s" << endl;
        }
    }
```

### 2. MemoryMonitor 클래스 (메모리 사용량 실시간 추적)

```cpp
class MemoryMonitor {
private:
    size_t initial_memory;
    size_t peak_memory;
    vector<size_t> memory_snapshots;
    vector<chrono::high_resolution_clock::time_point> timestamps;
    
    // 🧠 메모리 비교 결과
    struct MemoryComparison {
        size_t opencv_memory;        // OpenCV 메모리 사용량
        size_t custom_memory;        // Custom 메모리 사용량  
        double memory_ratio;         // 메모리 사용 비율
        size_t peak_difference;      // 최대 메모리 차이
        string efficiency_grade;     // 효율성 등급 (A~F)
    };
    
    map<string, MemoryComparison> comparisons;
    
public:
    // 📊 메모리 모니터링 시작
    void startMonitoring() {
        initial_memory = getCurrentMemoryUsage();
        peak_memory = initial_memory;
        memory_snapshots.clear();
        timestamps.clear();
        
        cout << "메모리 모니터링 시작 - 초기 메모리: " << initial_memory << " bytes" << endl;
    }
    
    // 📈 현재 메모리 사용량 확인
    size_t getCurrentMemoryUsage() {
#ifdef _WIN32
        PROCESS_MEMORY_COUNTERS pmc;
        if (GetProcessMemoryInfo(GetCurrentProcess(), &pmc, sizeof(pmc))) {
            return pmc.WorkingSetSize;
        }
#elif __linux__
        ifstream file("/proc/self/status");
        string line;
        while (getline(file, line)) {
            if (line.substr(0, 6) == "VmRSS:") {
                istringstream iss(line);
                string dummy;
                size_t memory_kb;
                iss >> dummy >> memory_kb;
                return memory_kb * 1024;  // KB to bytes
            }
        }
#endif
        return 0;
    }
    
    // 🎯 메모리 스냅샷 캡처
    void captureSnapshot() {
        size_t current = getCurrentMemoryUsage();
        memory_snapshots.push_back(current);
        timestamps.push_back(chrono::high_resolution_clock::now());
        
        if (current > peak_memory) {
            peak_memory = current;
        }
    }
    
    // 📊 메모리 효율성 비교
    MemoryComparison compareMemoryEfficiency(const string& algorithm_name,
                                           function<cv::Mat(const cv::Mat&)> opencv_func,
                                           function<cv::Mat(const cv::Mat&)> custom_func,
                                           const cv::Mat& test_image) {
        
        MemoryComparison result;
        
        // OpenCV 메모리 사용량 측정
        {
            startMonitoring();
            auto opencv_result = opencv_func(test_image);
            captureSnapshot();
            result.opencv_memory = peak_memory - initial_memory;
        }
        
        // Custom 메모리 사용량 측정
        {
            startMonitoring();  
            auto custom_result = custom_func(test_image);
            captureSnapshot();
            result.custom_memory = peak_memory - initial_memory;
        }
        
        // 비교 분석
        result.memory_ratio = static_cast<double>(result.custom_memory) / result.opencv_memory;
        result.peak_difference = abs(static_cast<long>(result.custom_memory - result.opencv_memory));
        
        // 효율성 등급 결정
        if (result.memory_ratio <= 1.1) result.efficiency_grade = "A";
        else if (result.memory_ratio <= 1.3) result.efficiency_grade = "B";
        else if (result.memory_ratio <= 1.5) result.efficiency_grade = "C";
        else if (result.memory_ratio <= 2.0) result.efficiency_grade = "D";
        else result.efficiency_grade = "F";
        
        comparisons[algorithm_name] = result;
        return result;
    }
    
    // 📋 메모리 리포트 생성
    void generateMemoryReport() {
        cout << "\n=== 메모리 사용량 분석 리포트 ===" << endl;
        cout << "┌─────────────────────┬──────────────┬──────────────┬─────────────┬──────────┐" << endl;
        cout << "│ 알고리즘            │ OpenCV (MB)  │ Custom (MB)  │ 비율        │ 등급     │" << endl;
        cout << "├─────────────────────┼──────────────┼──────────────┼─────────────┼──────────┤" << endl;
        
        for (const auto& [algorithm, comparison] : comparisons) {
            cout << "│ " << setw(19) << algorithm 
                 << " │ " << setw(12) << fixed << setprecision(2) << (comparison.opencv_memory / 1024.0 / 1024.0)
                 << " │ " << setw(12) << (comparison.custom_memory / 1024.0 / 1024.0)
                 << " │ " << setw(11) << comparison.memory_ratio << "x"
                 << " │ " << setw(8) << comparison.efficiency_grade
                 << " │" << endl;
        }
        cout << "└─────────────────────┴──────────────┴──────────────┴─────────────┴──────────┘" << endl;
        
        // 메모리 최적화 제안
        generateMemoryOptimizationTips();
    }
    
private:
    // 💡 메모리 최적화 팁 제시
    void generateMemoryOptimizationTips() {
        cout << "\n=== 메모리 최적화 제안 ===" << endl;
        
        for (const auto& [algorithm, comparison] : comparisons) {
            if (comparison.efficiency_grade == "D" || comparison.efficiency_grade == "F") {
                cout << "⚠️  " << algorithm << " 메모리 최적화 필요:" << endl;
                cout << "   🔧 불필요한 중간 변수 제거" << endl;
                cout << "   🔧 In-place 연산 활용" << endl;
                cout << "   🔧 메모리 풀링 사용" << endl;
                cout << "   🔧 스마트 포인터로 자동 메모리 관리" << endl;
            }
        }
        
        cout << "\n일반적 메모리 최적화 방법:" << endl;
        cout << "🔧 RAII 패턴으로 자동 리소스 관리" << endl;
        cout << "🔧 std::move 활용한 복사 오버헤드 제거" << endl;
        cout << "🔧 메모리 정렬을 통한 캐시 효율성 향상" << endl;
        cout << "🔧 큰 객체는 참조(&) 전달로 복사 비용 절약" << endl;
```

## ⚡ 실시간 성능 모니터링

### 3. RealTimeMonitor 클래스 (라이브 성능 추적)

```cpp
class RealTimeMonitor {
private:
    vector<double> performance_history;
    vector<double> accuracy_history;  
    vector<size_t> memory_history;
    vector<chrono::system_clock::time_point> time_stamps;
    
    // 📊 실시간 차트 데이터
    struct ChartData {
        vector<cv::Point2f> performance_points;
        vector<cv::Point2f> memory_points;
        cv::Scalar opencv_color = cv::Scalar(0, 255, 0);    // 녹색
        cv::Scalar custom_color = cv::Scalar(0, 0, 255);    // 빨간색
    };
    
    ChartData chart_data;
    bool monitoring_active = false;
    
public:
    // 🚀 실시간 모니터링 시작
    void startRealTimeMonitoring() {
        monitoring_active = true;
        performance_history.clear();
        accuracy_history.clear();
        memory_history.clear();
        time_stamps.clear();
        
        cout << "실시간 성능 모니터링 시작..." << endl;
        
        // 별도 스레드에서 모니터링 실행
        thread monitoring_thread(&RealTimeMonitor::monitoringLoop, this);
        monitoring_thread.detach();
    }
    
    // 📈 성능 데이터 업데이트
    void updatePerformanceData(const string& algorithm_name,
                              double opencv_time, 
                              double custom_time,
                              double accuracy,
                              size_t memory_usage) {
        
        if (!monitoring_active) return;
        
        // 성능 비율 계산
        double performance_ratio = custom_time / opencv_time;
        
        // 히스토리에 추가
        performance_history.push_back(performance_ratio);
        accuracy_history.push_back(accuracy);
        memory_history.push_back(memory_usage);
        time_stamps.push_back(chrono::system_clock::now());
        
        // 차트 데이터 업데이트
        updateChartData();
        
        // 실시간 출력
        cout << "[" << algorithm_name << "] "
             << "성능비율: " << performance_ratio << "x, "
             << "정확도: " << accuracy << "%, "
             << "메모리: " << (memory_usage / 1024.0 / 1024.0) << "MB" << endl;
    }
    
    // 📊 실시간 성능 차트 표시
    void showRealTimeChart() {
        if (performance_history.empty()) return;
        
        // 차트 이미지 생성 (800x600)
        cv::Mat chart(600, 800, CV_8UC3, cv::Scalar(255, 255, 255));
        
        // 성능 그래프 그리기
        drawPerformanceGraph(chart);
        
        // 메모리 그래프 그리기  
        drawMemoryGraph(chart);
        
        // 범례 및 정보 표시
        drawLegend(chart);
        drawStatistics(chart);
        
        // 차트 업데이트 표시
        cv::imshow("실시간 성능 모니터", chart);
        cv::waitKey(1);  // 1ms 대기로 실시간 업데이트
    }
    
    // 🔄 라이브 벤치마크 모드
    void runLiveBenchmark() {
        cout << "=== 라이브 벤치마크 모드 ===" << endl;
        cout << "ESC 키를 눌러 종료하세요." << endl;
        
        startRealTimeMonitoring();
        
        // 테스트 이미지 생성
        cv::Mat testImage = generateRandomTestImage();
        
        while (true) {
            // 랜덤 알고리즘 선택 및 실행
            string algorithm = selectRandomAlgorithm();
            runLiveTest(algorithm, testImage);
            
            // 실시간 차트 업데이트
            showRealTimeChart();
            
            // ESC 키 체크
            if (cv::waitKey(100) == 27) break;  // ESC
            
            // 1초 대기
            this_thread::sleep_for(chrono::milliseconds(1000));
        }
        
        stopMonitoring();
    }
    
private:
    // 🔄 모니터링 루프 (별도 스레드)
    void monitoringLoop() {
        while (monitoring_active) {
            // CPU 사용률 모니터링
            double cpu_usage = getCurrentCPUUsage();
            
            // 메모리 사용률 모니터링  
            size_t memory_usage = getCurrentMemoryUsage();
            
            // 시스템 성능 지표 수집
            collectSystemMetrics();
            
            this_thread::sleep_for(chrono::milliseconds(500));  // 0.5초마다 업데이트
        }
    }
    
    // 📊 차트 데이터 업데이트
    void updateChartData() {
        if (performance_history.empty()) return;
        
        chart_data.performance_points.clear();
        chart_data.memory_points.clear();
        
        int chart_width = 700;  // 차트 영역 너비
        int chart_height = 200; // 차트 영역 높이
        
        // 성능 데이터 정규화 및 포인트 생성
        double max_performance = *max_element(performance_history.begin(), performance_history.end());
        
        for (int i = 0; i < performance_history.size(); i++) {
            float x = (float)i / performance_history.size() * chart_width + 50;
            float y = chart_height - (performance_history[i] / max_performance * chart_height) + 50;
            chart_data.performance_points.push_back(cv::Point2f(x, y));
        }
        
        // 메모리 데이터 정규화 및 포인트 생성
        size_t max_memory = *max_element(memory_history.begin(), memory_history.end());
        
        for (int i = 0; i < memory_history.size(); i++) {
            float x = (float)i / memory_history.size() * chart_width + 50;
            float y = chart_height - ((float)memory_history[i] / max_memory * chart_height) + 350;
            chart_data.memory_points.push_back(cv::Point2f(x, y));
        }
    }
    
    // 📈 성능 그래프 그리기
    void drawPerformanceGraph(cv::Mat& chart) {
        if (chart_data.performance_points.size() < 2) return;
        
        // 성능 그래프 선 그리기
        for (int i = 1; i < chart_data.performance_points.size(); i++) {
            cv::line(chart, 
                    chart_data.performance_points[i-1], 
                    chart_data.performance_points[i], 
                    cv::Scalar(0, 0, 255), 2);  // 빨간색
        }
        
        // 축 및 라벨 그리기
        cv::line(chart, cv::Point(50, 50), cv::Point(50, 250), cv::Scalar(0, 0, 0), 2);      // Y축
        cv::line(chart, cv::Point(50, 250), cv::Point(750, 250), cv::Scalar(0, 0, 0), 2);    // X축
        
        cv::putText(chart, "Performance Ratio", cv::Point(60, 30), 
                   cv::FONT_HERSHEY_SIMPLEX, 0.7, cv::Scalar(0, 0, 0), 2);
    }
    
    // 🧠 메모리 그래프 그리기
    void drawMemoryGraph(cv::Mat& chart) {
        if (chart_data.memory_points.size() < 2) return;
        
        // 메모리 그래프 선 그리기
        for (int i = 1; i < chart_data.memory_points.size(); i++) {
            cv::line(chart, 
                    chart_data.memory_points[i-1], 
                    chart_data.memory_points[i], 
                    cv::Scalar(255, 0, 0), 2);  // 파란색
        }
        
        // 축 및 라벨 그리기
        cv::line(chart, cv::Point(50, 350), cv::Point(50, 550), cv::Scalar(0, 0, 0), 2);     // Y축
        cv::line(chart, cv::Point(50, 550), cv::Point(750, 550), cv::Scalar(0, 0, 0), 2);    // X축
        
        cv::putText(chart, "Memory Usage (MB)", cv::Point(60, 330), 
                   cv::FONT_HERSHEY_SIMPLEX, 0.7, cv::Scalar(0, 0, 0), 2);
    }
};
```
