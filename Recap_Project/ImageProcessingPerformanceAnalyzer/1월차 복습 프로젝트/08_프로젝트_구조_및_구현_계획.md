// 히스토그램 평활화 정확도 검증
void validateHistogramEqualization() {
    cv::Mat testImage = loadRealImage("data/test_images/grayscale/medium_gray.bmp");
    
    cv::Mat opencv_result = HistogramEqualization::opencv_version(testImage);
    cv::Mat custom_result = HistogramEqualization::custom_version(testImage);
    
    // 누적분포함수 검증
    auto opencv_hist = calculateHistogram(opencv_result);
    auto custom_hist = calculateHistogram(custom_result);
    
    double histogram_similarity = compareHistograms(opencv_hist, custom_hist);
    
    cout << "히스토그램 유사도: " << histogram_similarity << "%" << endl;
    assert(histogram_similarity > 95.0);  // 95% 이상 유사성 기대
}
