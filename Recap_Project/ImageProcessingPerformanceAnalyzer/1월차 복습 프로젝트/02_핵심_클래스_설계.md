# 🔧 핵심 클래스 설계

## 1. 성능 분석 및 측정 클래스

### PerformanceAnalyzer 클래스
```cpp
class PerformanceAnalyzer {
private:
    cv::TickMeter timer;
    
    struct BenchmarkResult {
        double opencvTime;           // OpenCV 함수 실행 시간
        double customTime;           // 직접 구현 실행 시간
        double speedupRatio;         // 성능 비율 (OpenCV 대비)
        double accuracy;             // 결과 정확도 (MSE 기반)
        string algorithmName;        // 알고리즘 이름
        int imageSize;              // 테스트 이미지 크기
        string imageType;           // 이미지 타입 (grayscale/color)
    };
    
    vector<BenchmarkResult> results;
    
public:
    // 성능 측정 래퍼 함수들
    template<typename Func>
    double measureExecutionTime(Func&& func, int iterations = 100);
    
    // 결과 정확도 비교
    double compareResults(const cv::Mat& opencv_result, const cv::Mat& custom_result);
    
    // 벤치마크 실행 및 결과 저장
    BenchmarkResult runComparison(const string& algorithm_name, 
                                  function<cv::Mat(const cv::Mat&)> opencv_func,
                                  function<cv::Mat(const cv::Mat&)> custom_func,
                                  const cv::Mat& input_image);
    
    // 결과 리포트 생성
    void generateReport(const string& output_file);
    void showRealTimeComparison();
};
```

class ComprehensiveBenchmark {

## 2. 이중 구현 알고리즘 클래스

### DualImplementationProcessor 클래스
```cpp
class DualImplementationProcessor {
public:
    // 1. 밝기 조절 비교
    struct BrightnessAdjustment {
        static cv::Mat opencv_version(const cv::Mat& src, int brightness);
        static cv::Mat custom_version(const cv::Mat& src, int brightness);
    };
    
    // 2. 명암비 조절 비교  
    struct ContrastAdjustment {
        static cv::Mat opencv_version(const cv::Mat& src, double alpha);
        static cv::Mat custom_version(const cv::Mat& src, double alpha);
    };
    
    // 3. 감마 보정 비교
    struct GammaCorrection {
        static cv::Mat opencv_version(const cv::Mat& src, double gamma);
        static cv::Mat custom_version(const cv::Mat& src, double gamma);
    };
    
    // 4. 히스토그램 계산 비교
    struct HistogramCalculation {
        static cv::Mat opencv_version(const cv::Mat& src);
        static vector<int> custom_version(const cv::Mat& src);
    };
    
    // 5. 히스토그램 평활화 비교
    struct HistogramEqualization {
        static cv::Mat opencv_version(const cv::Mat& src);
        static cv::Mat custom_version(const cv::Mat& src);
    };
    
    // 6. 색상 공간 변환 비교
    struct ColorSpaceConversion {
        static cv::Mat opencv_rgb_to_gray(const cv::Mat& src);
        static cv::Mat custom_rgb_to_gray(const cv::Mat& src);
        
        static cv::Mat opencv_rgb_to_hsv(const cv::Mat& src);
        static cv::Mat custom_rgb_to_hsv(const cv::Mat& src);
    };
    
    // 7. 영상 반전 비교
    struct ImageInversion {
        static cv::Mat opencv_version(const cv::Mat& src);
        static cv::Mat custom_version(const cv::Mat& src);
    };
};
```
