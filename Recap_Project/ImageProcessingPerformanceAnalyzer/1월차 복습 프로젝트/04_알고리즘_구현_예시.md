# 🔬 알고리즘 구현 예시 (OpenCV vs Custom)

## 1. 밝기 조절 (Brightness Adjustment)

### OpenCV 버전
```cpp
cv::Mat DualImplementationProcessor::BrightnessAdjustment::opencv_version(
    const cv::Mat& src, int brightness) {
    cv::Mat result;
    src.convertTo(result, -1, 1.0, brightness);  // alpha=1.0, beta=brightness
    return result;
}
```

### 직접 구현 버전 (최적화된)
```cpp
cv::Mat DualImplementationProcessor::BrightnessAdjustment::custom_version(
    const cv::Mat& src, int brightness) {
    cv::Mat result = src.clone();
    
    if (src.channels() == 1) {  // 그레이스케일
        for (int y = 0; y < src.rows; y++) {
            const uchar* src_ptr = src.ptr<uchar>(y);
            uchar* dst_ptr = result.ptr<uchar>(y);
            
            for (int x = 0; x < src.cols; x++) {
                int val = src_ptr[x] + brightness;
                dst_ptr[x] = cv::saturate_cast<uchar>(val);
            }
        }
    } else if (src.channels() == 3) {  // BGR 컬러
        for (int y = 0; y < src.rows; y++) {
            const cv::Vec3b* src_ptr = src.ptr<cv::Vec3b>(y);
            cv::Vec3b* dst_ptr = result.ptr<cv::Vec3b>(y);
            
            for (int x = 0; x < src.cols; x++) {
                for (int c = 0; c < 3; c++) {
                    int val = src_ptr[x][c] + brightness;
                    dst_ptr[x][c] = cv::saturate_cast<uchar>(val);
                }
            }
        }
    }
    return result;
}
```

### 성능 비교 실행 함수
```cpp
void compareBrightnessAdjustment(const cv::Mat& testImage, int brightness = 50) {

}
```

### OpenCV 버전 (방법 2: addWeighted 사용)
```cpp
cv::Mat DualImplementationProcessor::ContrastAdjustment::opencv_version_v2(

```

### 직접 구현 버전 (고급 명암비 조절)
```cpp
cv::Mat DualImplementationProcessor::ContrastAdjustment::custom_version(
    const cv::Mat& src, double alpha) {
    cv::Mat result = src.clone();
    
    if (src.channels() == 1) {  // 그레이스케일
        for (int y = 0; y < src.rows; y++) {
            const uchar* src_ptr = src.ptr<uchar>(y);
            uchar* dst_ptr = result.ptr<uchar>(y);
            
            for (int x = 0; x < src.cols; x++) {
                // 고급 명암비 조절: dst = src + (src - 128) * alpha
                int val = src_ptr[x] + (src_ptr[x] - 128) * alpha;
                dst_ptr[x] = cv::saturate_cast<uchar>(val);
            }
        }
    } else if (src.channels() == 3) {  // BGR 컬러
        for (int y = 0; y < src.rows; y++) {
            const cv::Vec3b* src_ptr = src.ptr<cv::Vec3b>(y);
            cv::Vec3b* dst_ptr = result.ptr<cv::Vec3b>(y);
            
            for (int x = 0; x < src.cols; x++) {
                for (int c = 0; c < 3; c++) {
                    int val = src_ptr[x][c] + (src_ptr[x][c] - 128) * alpha;
                    dst_ptr[x][c] = cv::saturate_cast<uchar>(val);
                }
            }
        }
    }
    return result;
}
```

## 3. 감마 보정 (Gamma Correction)

### OpenCV 버전 (LUT 사용)
```cpp
cv::Mat DualImplementationProcessor::GammaCorrection::opencv_version(
    const cv::Mat& src, double gamma) {
    cv::Mat lut(1, 256, CV_8U);
    uchar* lut_ptr = lut.ptr();
    
    for (int i = 0; i < 256; i++) {
        lut_ptr[i] = cv::saturate_cast<uchar>(pow(i / 255.0, 1.0 / gamma) * 255.0);
    }
    
    cv::Mat result;
    cv::LUT(src, lut, result);
    return result;
}
```

### 직접 구현 버전 (미리 계산된 LUT 사용)
```cpp
cv::Mat DualImplementationProcessor::GammaCorrection::custom_version(
    const cv::Mat& src, double gamma) {
    
    // 미리 계산된 감마 테이블 생성
    static double cached_gamma = -1.0;
    static uchar gamma_table[256];
    
    if (abs(gamma - cached_gamma) > 1e-6) {  // 감마 값이 변경된 경우만 재계산
        for (int i = 0; i < 256; i++) {
            gamma_table[i] = cv::saturate_cast<uchar>(
                255.0 * pow(i / 255.0, 1.0 / gamma) + 0.5
            );
        }
        cached_gamma = gamma;
    }
    
    cv::Mat result = src.clone();
    
    if (src.channels() == 1) {  // 그레이스케일
        for (int y = 0; y < src.rows; y++) {
            const uchar* src_ptr = src.ptr<uchar>(y);
            uchar* dst_ptr = result.ptr<uchar>(y);
            
            for (int x = 0; x < src.cols; x++) {
                dst_ptr[x] = gamma_table[src_ptr[x]];
            }
        }
    } else if (src.channels() == 3) {  // BGR 컬러
        for (int y = 0; y < src.rows; y++) {
            const cv::Vec3b* src_ptr = src.ptr<cv::Vec3b>(y);
            cv::Vec3b* dst_ptr = result.ptr<cv::Vec3b>(y);
            
            for (int x = 0; x < src.cols; x++) {
                for (int c = 0; c < 3; c++) {
                    dst_ptr[x][c] = gamma_table[src_ptr[x][c]];
                }
            }
        }
    }
    return result;
}
```

## 4. 히스토그램 계산 (Histogram Calculation)

### OpenCV 버전
```cpp
vector<int> DualImplementationProcessor::HistogramCalculation::opencv_version(
    const cv::Mat& src) {
    vector<cv::Mat> hist;
    vector<int> channels = {0};
    vector<int> histSize = {256};
    vector<float> ranges = {0, 256};
    
    cv::calcHist({src}, channels, cv::Mat(), hist, histSize, ranges);
    
    vector<int> result(256);
    for (int i = 0; i < 256; i++) {
        result[i] = static_cast<int>(hist[0].at<float>(i));
    }
    
    return result;
}
```

### 직접 구현 버전 (최적화된)
```cpp
vector<int> DualImplementationProcessor::HistogramCalculation::custom_version(
    const cv::Mat& src) {
    vector<int> histogram(256, 0);
    
    if (src.channels() == 1) {  // 그레이스케일
        for (int y = 0; y < src.rows; y++) {
            const uchar* ptr = src.ptr<uchar>(y);
            for (int x = 0; x < src.cols; x++) {
                histogram[ptr[x]]++;
            }
        }
    } else if (src.channels() == 3) {  // 컬러 → 그레이스케일 변환 후 히스토그램
        for (int y = 0; y < src.rows; y++) {
            const cv::Vec3b* ptr = src.ptr<cv::Vec3b>(y);
            for (int x = 0; x < src.cols; x++) {
                // RGB를 그레이스케일로 변환 (가중평균)
                int gray = static_cast<int>(
                    0.299 * ptr[x][2] +  // R
                    0.587 * ptr[x][1] +  // G  
                    0.114 * ptr[x][0]    // B
                );
                histogram[gray]++;
            }
        }
    }
    
    return histogram;
}
```

### 직접 구현 버전 (누적분포함수 활용)
```cpp
cv::Mat DualImplementationProcessor::HistogramEqualization::custom_version(
    const cv::Mat& src) {
    cv::Mat result = src.clone();
    
    // 1. 히스토그램 계산
    vector<int> histogram(256, 0);
    for (int y = 0; y < src.rows; y++) {
        const uchar* ptr = src.ptr<uchar>(y);
        for (int x = 0; x < src.cols; x++) {
            histogram[ptr[x]]++;
        }
    }
    
    // 2. 누적분포함수(CDF) 계산
    vector<float> cdf(256, 0);
    cdf[0] = histogram[0];
    for (int i = 1; i < 256; i++) {
        cdf[i] = cdf[i-1] + histogram[i];
    }
    
    // 3. CDF 정규화 (0~255 범위로)
    float total_pixels = src.rows * src.cols;
    vector<uchar> lut(256);
    for (int i = 0; i < 256; i++) {
        lut[i] = cv::saturate_cast<uchar>((cdf[i] / total_pixels) * 255.0f);
    }
    
    // 4. LUT 적용
    for (int y = 0; y < src.rows; y++) {
        const uchar* src_ptr = src.ptr<uchar>(y);
        uchar* dst_ptr = result.ptr<uchar>(y);
        for (int x = 0; x < src.cols; x++) {
            dst_ptr[x] = lut[src_ptr[x]];
        }
    }
    
    return result;
}
```

## 6. 색상 공간 변환 (Color Space Conversion)

### RGB → Grayscale 변환

#### OpenCV 버전
```cpp

    const cv::Mat& src) {
    cv::Mat result;
    cv::cvtColor(src, result, cv::COLOR_BGR2GRAY);
    return result;
}
```

#### 직접 구현 버전
```cpp
cv::Mat DualImplementationProcessor::ColorSpaceConversion::custom_rgb_to_gray(
    const cv::Mat& src) {
    cv::Mat result(src.rows, src.cols, CV_8UC1);
    
    for (int y = 0; y < src.rows; y++) {
        const cv::Vec3b* src_ptr = src.ptr<cv::Vec3b>(y);
        uchar* dst_ptr = result.ptr<uchar>(y);
        
        for (int x = 0; x < src.cols; x++) {
            // ITU-R BT.709 표준 가중치 사용
            float gray = 0.299f * src_ptr[x][2] +   // R
                        0.587f * src_ptr[x][1] +   // G
                        0.114f * src_ptr[x][0];    // B (OpenCV는 BGR 순서)
            dst_ptr[x] = cv::saturate_cast<uchar>(gray);
        }
    }
    
    return result;
}
```

```cpp
cv::Mat DualImplementationProcessor::ColorSpaceConversion::custom_rgb_to_hsv(
    const cv::Mat& src) {
    cv::Mat result(src.rows, src.cols, CV_8UC3);
    
    for (int y = 0; y < src.rows; y++) {
        const cv::Vec3b* src_ptr = src.ptr<cv::Vec3b>(y);
        cv::Vec3b* dst_ptr = result.ptr<cv::Vec3b>(y);
        
        for (int x = 0; x < src.cols; x++) {
            float b = src_ptr[x][0] / 255.0f;  // Blue
            float g = src_ptr[x][1] / 255.0f;  // Green  
            float r = src_ptr[x][2] / 255.0f;  // Red
            
            float max_val = max({r, g, b});
            float min_val = min({r, g, b});
            float delta = max_val - min_val;
            
            // Hue 계산
            float h = 0;
            if (delta != 0) {
                if (max_val == r) {
                    h = 60 * fmod((g - b) / delta, 6);
                } else if (max_val == g) {
                    h = 60 * ((b - r) / delta + 2);
                } else if (max_val == b) {
                    h = 60 * ((r - g) / delta + 4);
                }
            }
            if (h < 0) h += 360;
            
            // Saturation 계산
            float s = (max_val == 0) ? 0 : delta / max_val;
            
            // Value 계산
            float v = max_val;
            
            // OpenCV HSV 범위로 변환 (H: 0-179, S: 0-255, V: 0-255)
            dst_ptr[x][0] = cv::saturate_cast<uchar>(h / 2);     // H
            dst_ptr[x][1] = cv::saturate_cast<uchar>(s * 255);   // S
            dst_ptr[x][2] = cv::saturate_cast<uchar>(v * 255);   // V
        }
    }
    
    return result;
}
```

## 7. 영상 반전 (Image Inversion)

### OpenCV 버전
```cpp
cv::Mat DualImplementationProcessor::ImageInversion::opencv_version(
    const cv::Mat& src) {
    cv::Mat result;
    cv::bitwise_not(src, result);
    return result;
}
```

### 직접 구현 버전
```cpp
cv::Mat DualImplementationProcessor::ImageInversion::custom_version(
    const cv::Mat& src) {
    cv::Mat result = src.clone();
    
    if (src.channels() == 1) {  // 그레이스케일
        for (int y = 0; y < src.rows; y++) {
            const uchar* src_ptr = src.ptr<uchar>(y);
            uchar* dst_ptr = result.ptr<uchar>(y);
            
            for (int x = 0; x < src.cols; x++) {
                dst_ptr[x] = 255 - src_ptr[x];
            }
        }
    } else if (src.channels() == 3) {  // BGR 컬러
        for (int y = 0; y < src.rows; y++) {
            const cv::Vec3b* src_ptr = src.ptr<cv::Vec3b>(y);
            cv::Vec3b* dst_ptr = result.ptr<cv::Vec3b>(y);
            
            for (int x = 0; x < src.cols; x++) {
                for (int c = 0; c < 3; c++) {
                    dst_ptr[x][c] = 255 - src_ptr[x][c];
                }
            }
        }
    }
    return result;
}
```
