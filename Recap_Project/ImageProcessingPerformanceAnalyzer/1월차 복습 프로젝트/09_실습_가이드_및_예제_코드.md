### 1. 개발 환경 설정

#### **필수 요구사항**
- **C++ 컴파일러**: Visual Studio 2019+ (Windows) 또는 GCC 9+ (Linux/Mac)
- **OpenCV 4.x**: 최신 버전 권장
- **CMake 3.16+**: 크로스 플랫폼 빌드 지원

#### **OpenCV 설치 (Windows)**
```bash
# vcpkg 사용 (권장)
git clone https://github.com/Microsoft/vcpkg.git
cd vcpkg
./bootstrap-vcpkg.bat
./vcpkg integrate install
./vcpkg install opencv[contrib]:x64-windows
```

#### **프로젝트 생성 및 빌드**
```bash

mkdir ImageProcessingPerformanceAnalyzer
cd ImageProcessingPerformanceAnalyzer

# CMakeLists.txt 생성 (아래 예제 참조)
cmake -B build -S . -DCMAKE_TOOLCHAIN_FILE=[vcpkg root]/scripts/buildsystems/vcpkg.cmake
cmake --build build --config Release
```

### 3. 상세 구현 예제

#### **PerformanceAnalyzer.cpp - 성능 측정 엔진**
```cpp
#include "PerformanceAnalyzer.h"
#include <opencv2/opencv.hpp>
#include <fstream>
#include <iomanip>

using namespace cv;
using namespace std;

// 결과 정확도 비교 (다중 지표)
PerformanceAnalyzer::AccuracyMetrics PerformanceAnalyzer::compareResults(
    const Mat& opencv_result, const Mat& custom_result) {
    
    AccuracyMetrics metrics;
    
    // 1. MSE (Mean Squared Error) 계산
    Mat diff;
    absdiff(opencv_result, custom_result, diff);
    diff.convertTo(diff, CV_32F);
    
    Scalar mse_scalar = mean(diff.mul(diff));
    metrics.mse = mse_scalar[0];
    
    // 2. PSNR (Peak Signal-to-Noise Ratio) 계산
    if (metrics.mse > 0) {
        metrics.psnr = 20.0 * log10(255.0 / sqrt(metrics.mse));
    } else {
        metrics.psnr = 100.0;  // 완벽히 동일한 경우
    }
    
    // 3. SSIM (Structural Similarity Index) 계산
    metrics.ssim = calculateSSIM(opencv_result, custom_result);
    
    // 4. 히스토그램 유사도 (그레이스케일인 경우)
    if (opencv_result.channels() == 1) {
        vector<Mat> opencv_imgs = {opencv_result};
        vector<Mat> custom_imgs = {custom_result};
        
        Mat opencv_hist, custom_hist;
        int histSize = 256;
        float range[] = {0, 256};
        const float* histRange = {range};
        
        calcHist(opencv_imgs, {0}, Mat(), opencv_hist, {histSize}, {histRange});
        calcHist(custom_imgs, {0}, Mat(), custom_hist, {histSize}, {histRange});
        
        // 정규화
        normalize(opencv_hist, opencv_hist, 0, 1, NORM_MINMAX, -1, Mat());
        normalize(custom_hist, custom_hist, 0, 1, NORM_MINMAX, -1, Mat());
        
        // 상관계수 계산
        metrics.histogram_similarity = compareHist(opencv_hist, custom_hist, HISTCMP_CORREL) * 100.0;
    } else {
        metrics.histogram_similarity = 100.0;  // 컬러 이미지는 100%로 가정
    }
    
    return metrics;
}

// SSIM 계산 (단순화 버전)
double PerformanceAnalyzer::calculateSSIM(const Mat& img1, const Mat& img2) {
    Mat img1_32f, img2_32f;
    img1.convertTo(img1_32f, CV_32F);
    img2.convertTo(img2_32f, CV_32F);
    
    // 평균 계산
    Scalar mu1 = mean(img1_32f);
    Scalar mu2 = mean(img2_32f);
    
    // 분산 및 공분산 계산 (단순화)
    Mat img1_sq, img2_sq, img1_img2;
    multiply(img1_32f, img1_32f, img1_sq);
    multiply(img2_32f, img2_32f, img2_sq);
    multiply(img1_32f, img2_32f, img1_img2);
    
    Scalar sigma1_sq = mean(img1_sq) - Scalar(mu1[0] * mu1[0]);
    Scalar sigma2_sq = mean(img2_sq) - Scalar(mu2[0] * mu2[0]);
    Scalar sigma12 = mean(img1_img2) - Scalar(mu1[0] * mu2[0]);
    
    // SSIM 상수
    double C1 = (0.01 * 255) * (0.01 * 255);
    double C2 = (0.03 * 255) * (0.03 * 255);
    
    // SSIM 계산
    double numerator = (2 * mu1[0] * mu2[0] + C1) * (2 * sigma12[0] + C2);
    double denominator = (mu1[0] * mu1[0] + mu2[0] * mu2[0] + C1) * (sigma1_sq[0] + sigma2_sq[0] + C2);
    
    return numerator / denominator;
}

// 벤치마크 실행 및 결과 저장
BenchmarkResult PerformanceAnalyzer::runComparison(
    const string& algorithm_name,
    function<Mat(const Mat&)> opencv_func,
    function<Mat(const Mat&)> custom_func,
    const Mat& input_image,
    int iterations) {
    
    BenchmarkResult result;
    result.algorithmName = algorithm_name;
    result.imageSize = input_image.rows * input_image.cols;
    result.imageType = (input_image.channels() == 1) ? "grayscale" : "color";
    
    Mat opencv_result, custom_result;
    
    // OpenCV 함수 성능 측정
    result.opencvTime = measureExecutionTime([&]() {
        opencv_result = opencv_func(input_image);
    }, iterations);
    
    // Custom 함수 성능 측정
    result.customTime = measureExecutionTime([&]() {
        custom_result = custom_func(input_image);
    }, iterations);
    
    // 성능 비율 계산
    result.speedupRatio = result.customTime / result.opencvTime;
    
    // 정확도 분석
    auto accuracy_metrics = compareResults(opencv_result, custom_result);
    
    // 종합 정확도 점수 (가중 평균)
    result.accuracy = (accuracy_metrics.ssim * 100 * 0.4) +      // SSIM 40%
                     (min(accuracy_metrics.psnr / 40.0, 1.0) * 100 * 0.3) +  // PSNR 30%
                     (accuracy_metrics.histogram_similarity * 0.3);            // Histogram 30%
    
    // 메모리 사용량 측정 (단순화)
    result.memoryUsage = input_image.total() * input_image.elemSize();
    
    // 결과 저장
    results.push_back(result);
    
    return result;
}

// XML 리포트 생성
void PerformanceAnalyzer::generateReport(const string& output_file) {
    FileStorage fs(output_file, FileStorage::WRITE);
    
    fs << "PerformanceAnalysisReport" << "{";
    fs << "GeneratedDate" << getCurrentDateTime();
    fs << "TotalAlgorithms" << static_cast<int>(results.size());
    
    fs << "Results" << "[";
    for (const auto& result : results) {
        fs << "{";
        fs << "Algorithm" << result.algorithmName;
        fs << "OpenCVTime" << result.opencvTime;
        fs << "CustomTime" << result.customTime;
        fs << "PerformanceRatio" << result.speedupRatio;
        fs << "Accuracy" << result.accuracy;
        fs << "ImageSize" << result.imageSize;
        fs << "ImageType" << result.imageType;
        fs << "MemoryUsage" << static_cast<int>(result.memoryUsage);
        fs << "}";
    }
    fs << "]";
    
    fs << "}";
    fs.release();
    
    cout << "📊 리포트 생성 완료: " << output_file << endl;
}

// 현재 날짜/시간 문자열 반환
string PerformanceAnalyzer::getCurrentDateTime() {
    auto now = chrono::system_clock::now();
    auto time_t = chrono::system_clock::to_time_t(now);
    auto tm = *localtime(&time_t);
    
    stringstream ss;
    ss << put_time(&tm, "%Y-%m-%d %H:%M:%S");
    return ss.str();
}
```

### 4. 실시간 시각화 예제

#### **RealTimeDemo.cpp - 실시간 모니터링 데모**
```cpp
#include "RealTimeDemo.h"
#include <thread>
#include <chrono>

void runRealTimeDemo() {
    cout << "\n=== 실시간 성능 모니터링 데모 ===" << endl;
    cout << "ESC 키를 눌러 종료하세요." << endl;
    
    RealTimeMonitor monitor;
    PerformanceAnalyzer analyzer;
    TestImageGenerator generator;
    
    // 모니터링 시작
    monitor.startRealTimeMonitoring();
    
    // 테스트 이미지들 준비
    vector<Mat> testImages = {
        generator.createTestImage(Size(256, 256), CV_8UC1),
        generator.createTestImage(Size(512, 512), CV_8UC1),
        generator.createTestImage(Size(256, 256), CV_8UC3),
        generator.createTestImage(Size(512, 512), CV_8UC3)
    };
    
    // 알고리즘 목록
    vector<string> algorithms = {
        "BrightnessAdjustment",
        "ContrastAdjustment", 
        "GammaCorrection",
        "ImageInversion"
    };
    
    int iteration = 0;
    
    while (true) {
        // 랜덤 선택
        Mat current_image = testImages[iteration % testImages.size()];
        string current_algorithm = algorithms[iteration % algorithms.size()];
        
        cout << "\n[" << iteration + 1 << "] " << current_algorithm 
             << " - 이미지 크기: " << current_image.size() << endl;
        
        // 성능 측정
        BenchmarkResult result;
        
        if (current_algorithm == "BrightnessAdjustment") {
            result = testBrightnessAdjustment(current_image, analyzer);
        } else if (current_algorithm == "GammaCorrection") {
            result = testGammaCorrection(current_image, analyzer);
        }
        // ... 다른 알고리즘들
        
        // 실시간 데이터 업데이트
        monitor.updatePerformanceData(
            current_algorithm,
            result.opencvTime,
            result.customTime,
            result.accuracy,
            result.memoryUsage
        );
        
        // 실시간 차트 표시
        monitor.showRealTimeChart();
        
        // 사용자 입력 확인
        if (waitKey(100) == 27) break;  // ESC
        
        // 다음 반복 준비
        iteration++;
        this_thread::sleep_for(chrono::milliseconds(500));
        
        // 10회마다 상세 분석 표시
        if (iteration % 10 == 0) {
            showDetailedAnalysis(analyzer);
        }
    }
    
    monitor.stopMonitoring();
    cout << "\n✅ 실시간 모니터링 종료" << endl;
}

// 상세 분석 표시
void showDetailedAnalysis(const PerformanceAnalyzer& analyzer) {
    cout << "\n📊 현재까지의 분석 결과:" << endl;
    
    const auto& results = analyzer.getResults();
    if (results.empty()) return;
    
    // 평균 성능 계산
    double avg_performance_ratio = 0;
    double avg_accuracy = 0;
    
    for (const auto& result : results) {
        avg_performance_ratio += result.speedupRatio;
        avg_accuracy += result.accuracy;
    }
    
    avg_performance_ratio /= results.size();
    avg_accuracy /= results.size();
    
    cout << "   평균 성능 비율: " << fixed << setprecision(2) << avg_performance_ratio << "x" << endl;
    cout << "   평균 정확도: " << avg_accuracy << "%" << endl;
    cout << "   총 테스트 수: " << results.size() << endl;
    
    // 최고/최저 성능 표시
    auto max_perf = max_element(results.begin(), results.end(),
        [](const BenchmarkResult& a, const BenchmarkResult& b) {
            return a.speedupRatio < b.speedupRatio;
        });
        
    auto min_perf = min_element(results.begin(), results.end(),
        [](const BenchmarkResult& a, const BenchmarkResult& b) {
            return a.speedupRatio < b.speedupRatio;
        });
    
    cout << "   최대 성능 차이: " << max_perf->algorithmName << " (" << max_perf->speedupRatio << "x)" << endl;
    cout << "   최소 성능 차이: " << min_perf->algorithmName << " (" << min_perf->speedupRatio << "x)" << endl;
}
```

```cpp
// 밝기 조절 트랙바 콜백
void onBrightnessChange(int brightness, void* userdata) {
    if (g_original_image.empty()) return;
    
    int brightness_value = brightness - 100;  // -100 ~ +100 범위
    
    // 성능 측정과 함께 처리
    TickMeter timer;
    
    // OpenCV 버전
    timer.start();
    g_opencv_result = DualImplementationProcessor::BrightnessAdjustment::opencv_version(
        g_original_image, brightness_value);
    timer.stop();
    double opencv_time = timer.getTimeMilli();
    
    // Custom 버전
    timer.reset();
    timer.start();
    g_custom_result = DualImplementationProcessor::BrightnessAdjustment::custom_version(
        g_original_image, brightness_value);
    timer.stop();
    double custom_time = timer.getTimeMilli();
    
    // 결과 표시
    imshow("OpenCV Result", g_opencv_result);
    imshow("Custom Result", g_custom_result);
    
    // 성능 정보 오버레이
    Mat info_overlay = g_custom_result.clone();
    if (info_overlay.channels() == 1) {
        cvtColor(info_overlay, info_overlay, COLOR_GRAY2BGR);
    }
    
    // 성능 정보 텍스트
    string opencv_info = "OpenCV: " + to_string(opencv_time).substr(0, 5) + "ms";
    string custom_info = "Custom: " + to_string(custom_time).substr(0, 5) + "ms";
    string ratio_info = "Ratio: " + to_string(custom_time / opencv_time).substr(0, 4) + "x";
    
    putText(info_overlay, opencv_info, Point(10, 30), FONT_HERSHEY_SIMPLEX, 0.6, Scalar(0, 255, 0), 2);
    putText(info_overlay, custom_info, Point(10, 60), FONT_HERSHEY_SIMPLEX, 0.6, Scalar(0, 0, 255), 2);
    putText(info_overlay, ratio_info, Point(10, 90), FONT_HERSHEY_SIMPLEX, 0.6, Scalar(255, 0, 0), 2);
    
    namedWindow("Performance Info", WINDOW_AUTOSIZE);
    imshow("Performance Info", info_overlay);
}

// 빠른 벤치마크 실행
void runQuickBenchmark() {
    cout << "\n🚀 빠른 벤치마크 실행 중..." << endl;
    
    vector<int> test_params = {-50, -25, 0, 25, 50, 75, 100};
    vector<double> opencv_times, custom_times;
    
    for (int brightness : test_params) {
        // OpenCV 측정
        auto opencv_func = [brightness](const Mat& img) {
            return DualImplementationProcessor::BrightnessAdjustment::opencv_version(img, brightness);
        };
        double opencv_time = g_analyzer.measureExecutionTime(
            [&]() { opencv_func(g_original_image); }, 50);
        
        // Custom 측정
        auto custom_func = [brightness](const Mat& img) {
            return DualImplementationProcessor::BrightnessAdjustment::custom_version(img, brightness);
        };
        double custom_time = g_analyzer.measureExecutionTime(
            [&]() { custom_func(g_original_image); }, 50);
        
        opencv_times.push_back(opencv_time);
        custom_times.push_back(custom_time);
        
        cout << "   밝기 " << brightness << ": OpenCV=" << opencv_time 
             << "ms, Custom=" << custom_time << "ms" << endl;
    }
    
    // 평균 성능 계산
    double avg_opencv = accumulate(opencv_times.begin(), opencv_times.end(), 0.0) / opencv_times.size();
    double avg_custom = accumulate(custom_times.begin(), custom_times.end(), 0.0) / custom_times.size();
    
    cout << "📊 평균 성능:" << endl;
    cout << "   OpenCV: " << avg_opencv << "ms" << endl;
    cout << "   Custom: " << avg_custom << "ms" << endl;
    cout << "   성능 비율: " << (avg_custom / avg_opencv) << "x" << endl;
}

    if (!g_opencv_result.empty() && !g_custom_result.empty()) {
```

```cpp
void comparePixelAccessMethods(const Mat& image) {
    cout << "\n--- 픽셀 접근 방법 비교 ---" << endl;
    
    TickMeter timer;
    Mat result;
    
    // 방법 1: at() 함수 (느림)
    timer.start();
    result = image.clone();
    for (int y = 0; y < image.rows; y++) {
        for (int x = 0; x < image.cols; x++) {
            result.at<uchar>(y, x) = saturate_cast<uchar>(image.at<uchar>(y, x) + 50);
        }
    }
    timer.stop();
    cout << "at() 방법: " << timer.getTimeMilli() << "ms" << endl;
    
    // 방법 2: ptr() 함수 (빠름)
    timer.reset();
    timer.start();
    result = image.clone();
    for (int y = 0; y < image.rows; y++) {
        const uchar* src_ptr = image.ptr<uchar>(y);
        uchar* dst_ptr = result.ptr<uchar>(y);
        for (int x = 0; x < image.cols; x++) {
            dst_ptr[x] = saturate_cast<uchar>(src_ptr[x] + 50);
        }
    }
    timer.stop();
    cout << "ptr() 방법: " << timer.getTimeMilli() << "ms" << endl;
    
    // 방법 3: 연속 메모리 접근 (가장 빠름)
    timer.reset();
    timer.start();
    result = image.clone();
    if (image.isContinuous() && result.isContinuous()) {
        const uchar* src_data = image.ptr<uchar>(0);
        uchar* dst_data = result.ptr<uchar>(0);
        int total_pixels = image.rows * image.cols;
        
        for (int i = 0; i < total_pixels; i++) {
            dst_data[i] = saturate_cast<uchar>(src_data[i] + 50);
        }
    }
    timer.stop();
    cout << "연속 접근: " << timer.getTimeMilli() << "ms" << endl;
}
```
