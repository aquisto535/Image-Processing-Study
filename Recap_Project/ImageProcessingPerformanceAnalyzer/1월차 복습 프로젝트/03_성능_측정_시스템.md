# ⏱️ 성능 측정 시스템

## 1. TickMeter 기반 정밀 성능 측정

### 기본 성능 측정 구조
```cpp
template<typename Func>
double PerformanceAnalyzer::measureExecutionTime(Func&& func, int iterations) {
    cv::TickMeter timer;
    
    // 캐시 워밍업 (첫 실행 시 캐시 미스로 인한 오차 제거)
    func();
    
    timer.start();
    for (int i = 0; i < iterations; i++) {
        func();
    }
    timer.stop();
    
    return timer.getTimeMilli() / iterations;  // 평균 실행 시간 반환
}
```

### 다중 반복 측정으로 정확도 향상
```cpp
struct DetailedMeasurement {
    double min_time;
    double max_time;
    double avg_time;
    double std_deviation;
    vector<double> all_measurements;
};

DetailedMeasurement PerformanceAnalyzer::measureDetailed(
    function<void()> func, int iterations, int rounds) {
    
    vector<double> round_averages;
    
    for (int round = 0; round < rounds; round++) {
        cv::TickMeter timer;
        timer.start();
        
        for (int i = 0; i < iterations; i++) {
            func();
        }
        
        timer.stop();
        round_averages.push_back(timer.getTimeMilli() / iterations);
    }
    
    DetailedMeasurement result;
    result.all_measurements = round_averages;
    result.min_time = *min_element(round_averages.begin(), round_averages.end());
    result.max_time = *max_element(round_averages.begin(), round_averages.end());
    result.avg_time = accumulate(round_averages.begin(), round_averages.end(), 0.0) / rounds;
    
    // 표준편차 계산
    double variance = 0.0;
    for (double time : round_averages) {
        variance += pow(time - result.avg_time, 2);
    }
    result.std_deviation = sqrt(variance / rounds);
    
    return result;
}
```

```cpp
class WindowsMemoryMonitor {
private:
    PROCESS_MEMORY_COUNTERS_EX initial_memory;
    PROCESS_MEMORY_COUNTERS_EX peak_memory;
    HANDLE process_handle;
    
public:
    WindowsMemoryMonitor() {
        process_handle = GetCurrentProcess();
        GetProcessMemoryInfo(process_handle, 
                           (PROCESS_MEMORY_COUNTERS*)&initial_memory, 
                           sizeof(initial_memory));
        peak_memory = initial_memory;
    }
    
    size_t getCurrentMemoryUsage() {
        PROCESS_MEMORY_COUNTERS_EX current_memory;
        GetProcessMemoryInfo(process_handle, 
                           (PROCESS_MEMORY_COUNTERS*)&current_memory, 
                           sizeof(current_memory));
        
        if (current_memory.WorkingSetSize > peak_memory.WorkingSetSize) {
            peak_memory = current_memory;
        }
        
        return current_memory.WorkingSetSize;
    }
    
    size_t getPeakMemoryUsage() {
        return peak_memory.WorkingSetSize;
    }
    
    size_t getMemoryDifference() {
        return getCurrentMemoryUsage() - initial_memory.WorkingSetSize;
    }
};
```

### 크로스 플랫폼 메모리 모니터링
```cpp
class CrossPlatformMemoryMonitor {
private:
    size_t initial_memory;
    size_t peak_memory;
    
    size_t getCurrentMemoryUsageImpl() {
#ifdef _WIN32
        PROCESS_MEMORY_COUNTERS_EX pmc;
        GetProcessMemoryInfo(GetCurrentProcess(), 
                           (PROCESS_MEMORY_COUNTERS*)&pmc, sizeof(pmc));
        return pmc.WorkingSetSize;
#elif defined(__linux__)
        long rss = 0L;
        FILE* file = fopen("/proc/self/statm", "r");
        if (file == NULL) return 0;
        if (fscanf(file, "%*s%ld", &rss) != 1) {
            fclose(file);
            return 0;
        }
        fclose(file);
        return rss * sysconf(_SC_PAGESIZE);
#elif defined(__APPLE__)
        struct task_info t_info;
        mach_msg_type_number_t t_info_count = TASK_INFO_MAX;
        if (task_info(mach_task_self(), TASK_BASIC_INFO, 
                     (task_info_t)&t_info, &t_info_count) != KERN_SUCCESS)
            return 0;
        return t_info.resident_size;
#else
        return 0;  // Unsupported platform
#endif
    }
    
public:
    void startMonitoring() {
        initial_memory = getCurrentMemoryUsageImpl();
        peak_memory = initial_memory;
    }
    
    size_t getCurrentMemoryUsage() {
        size_t current = getCurrentMemoryUsageImpl();
        if (current > peak_memory) {
            peak_memory = current;
        }
        return current;
    }
};
```

## 3. 정확도 측정 알고리즘

### MSE (Mean Squared Error) 계산
```cpp
double AccuracyAnalyzer::calculateMSE(const cv::Mat& img1, const cv::Mat& img2) {
    if (img1.size() != img2.size() || img1.type() != img2.type()) {
        throw invalid_argument("Images must have same size and type");
    }
    
    cv::Mat diff;
    cv::absdiff(img1, img2, diff);
    diff.convertTo(diff, CV_64F);
    
    cv::Scalar mse_scalar = cv::mean(diff.mul(diff));
    
    // 채널별 MSE 평균 계산
    double mse = 0.0;
    for (int i = 0; i < diff.channels(); i++) {
        mse += mse_scalar[i];
    }
    
    return mse / diff.channels();
}
```

### PSNR (Peak Signal-to-Noise Ratio) 계산
```cpp
double AccuracyAnalyzer::calculatePSNR(const cv::Mat& img1, const cv::Mat& img2) {
    double mse = calculateMSE(img1, img2);
    
    if (mse == 0.0) {
        return numeric_limits<double>::infinity();  // 완전히 동일한 이미지
    }
    
    double max_pixel_value = 255.0;  // 8비트 이미지 기준
    return 10.0 * log10((max_pixel_value * max_pixel_value) / mse);
}
```

### SSIM (Structural Similarity Index) 계산
```cpp
double AccuracyAnalyzer::calculateSSIM(const cv::Mat& img1, const cv::Mat& img2) {
    const double C1 = 6.5025;    // (K1*L)^2, K1=0.01, L=255
    const double C2 = 58.5225;   // (K2*L)^2, K2=0.03, L=255
    
    cv::Mat I1, I2;
    img1.convertTo(I1, CV_64F);
    img2.convertTo(I2, CV_64F);
    
    cv::Mat I1_sq = I1.mul(I1);
    cv::Mat I2_sq = I2.mul(I2);
    cv::Mat I1_I2 = I1.mul(I2);
    
    cv::Mat mu1, mu2;
    cv::GaussianBlur(I1, mu1, cv::Size(11, 11), 1.5);
    cv::GaussianBlur(I2, mu2, cv::Size(11, 11), 1.5);
    
    cv::Mat mu1_sq = mu1.mul(mu1);
    cv::Mat mu2_sq = mu2.mul(mu2);
    cv::Mat mu1_mu2 = mu1.mul(mu2);
    
    cv::Mat sigma1_sq, sigma2_sq, sigma12;
    cv::GaussianBlur(I1_sq, sigma1_sq, cv::Size(11, 11), 1.5);
    sigma1_sq -= mu1_sq;
    
    cv::GaussianBlur(I2_sq, sigma2_sq, cv::Size(11, 11), 1.5);
    sigma2_sq -= mu2_sq;
    
    cv::GaussianBlur(I1_I2, sigma12, cv::Size(11, 11), 1.5);
    sigma12 -= mu1_mu2;
    
    cv::Mat numerator = (2 * mu1_mu2 + C1).mul(2 * sigma12 + C2);
    cv::Mat denominator = (mu1_sq + mu2_sq + C1).mul(sigma1_sq + sigma2_sq + C2);
    
    cv::Mat ssim_map;
    cv::divide(numerator, denominator, ssim_map);
    
    cv::Scalar mean_ssim = cv::mean(ssim_map);
    
    // 채널별 SSIM 평균
    double ssim = 0.0;
    for (int i = 0; i < ssim_map.channels(); i++) {
        ssim += mean_ssim[i];
    }
    
    return ssim / ssim_map.channels();
}
```

## 4. 히스토그램 유사도 측정

### 히스토그램 비교 지표들
```cpp
double AccuracyAnalyzer::compareHistograms(const vector<int>& hist1, 
                                         const vector<int>& hist2) {
    if (hist1.size() != hist2.size()) {
        throw invalid_argument("Histograms must have same size");
    }
    
    // 1. 코사인 유사도 (Cosine Similarity)
    double dot_product = 0.0;
    double norm1 = 0.0, norm2 = 0.0;
    
    for (size_t i = 0; i < hist1.size(); i++) {
        dot_product += hist1[i] * hist2[i];
        norm1 += hist1[i] * hist1[i];
        norm2 += hist2[i] * hist2[i];
    }
    
    if (norm1 == 0.0 || norm2 == 0.0) return 0.0;
    
    double cosine_similarity = dot_product / (sqrt(norm1) * sqrt(norm2));
    
    // 2. 카이제곱 거리 (Chi-Square Distance)
    double chi_square = 0.0;
    for (size_t i = 0; i < hist1.size(); i++) {
        if (hist1[i] + hist2[i] != 0) {
            double diff = hist1[i] - hist2[i];
            chi_square += (diff * diff) / (hist1[i] + hist2[i]);
        }
    }
    
    // 3. 바타차야 거리 (Bhattacharyya Distance)
    double bhattacharyya = 0.0;
    double sum1 = accumulate(hist1.begin(), hist1.end(), 0.0);
    double sum2 = accumulate(hist2.begin(), hist2.end(), 0.0);
    
    for (size_t i = 0; i < hist1.size(); i++) {
        double p1 = hist1[i] / sum1;
        double p2 = hist2[i] / sum2;
        bhattacharyya += sqrt(p1 * p2);
    }
    
    bhattacharyya = -log(bhattacharyya);
    
    // 종합 점수 (코사인 유사도 위주, 다른 거리들로 보정)
    double similarity_score = cosine_similarity * exp(-chi_square * 0.001) * exp(-bhattacharyya);
    
    return similarity_score;
}
```

## 6. 통계적 분석

### 성능 분포 분석
```cpp
struct PerformanceStatistics {
    double mean;
    double median;
    double std_deviation;
    double confidence_interval_95_lower;
    double confidence_interval_95_upper;
    double percentile_95;
    double percentile_99;
};

PerformanceStatistics PerformanceAnalyzer::calculateStatistics(
    const vector<double>& measurements) {
    
    PerformanceStatistics stats;
    
    // 정렬된 복사본 생성
    vector<double> sorted_data = measurements;
    sort(sorted_data.begin(), sorted_data.end());
    
    // 평균
    stats.mean = accumulate(sorted_data.begin(), sorted_data.end(), 0.0) / sorted_data.size();
    
    // 중앙값
    size_t n = sorted_data.size();
    if (n % 2 == 0) {
        stats.median = (sorted_data[n/2 - 1] + sorted_data[n/2]) / 2.0;
    } else {
        stats.median = sorted_data[n/2];
    }
    
    // 표준편차
    double variance = 0.0;
    for (double value : sorted_data) {
        variance += pow(value - stats.mean, 2);
    }
    stats.std_deviation = sqrt(variance / (n - 1));
    
    // 95% 신뢰구간 (t-분포 기반)
    double t_value = 1.96;  // 대략적인 값, 정확한 값은 n에 따라 달라짐
    double margin_error = t_value * (stats.std_deviation / sqrt(n));
    stats.confidence_interval_95_lower = stats.mean - margin_error;
    stats.confidence_interval_95_upper = stats.mean + margin_error;
    
    // 백분위수
    stats.percentile_95 = sorted_data[min((size_t)(n * 0.95), n - 1)];
    stats.percentile_99 = sorted_data[min((size_t)(n * 0.99), n - 1)];
    
    return stats;
}
```

### 결과 직렬화
```cpp
class BenchmarkResultSerializer {
public:
    static void saveToXML(const vector<BenchmarkResult>& results, 
                         const string& filename) {
        cv::FileStorage fs(filename, cv::FileStorage::WRITE);
        
        fs << "benchmark_results" << "[";
        for (const auto& result : results) {
            fs << "{";
            fs << "algorithm_name" << result.algorithmName;
            fs << "opencv_time" << result.opencvTime;
            fs << "custom_time" << result.customTime;
            fs << "speedup_ratio" << result.speedupRatio;
            fs << "accuracy" << result.accuracy;
            fs << "image_size" << result.imageSize;
            fs << "image_type" << result.imageType;
            fs << "}";
        }
        fs << "]";
        fs.release();
    }
    
    static vector<BenchmarkResult> loadFromXML(const string& filename) {
        vector<BenchmarkResult> results;
        cv::FileStorage fs(filename, cv::FileStorage::READ);
        
        cv::FileNode benchmark_node = fs["benchmark_results"];
        for (cv::FileNodeIterator it = benchmark_node.begin(); 
             it != benchmark_node.end(); ++it) {
            
            BenchmarkResult result;
            (*it)["algorithm_name"] >> result.algorithmName;
            (*it)["opencv_time"] >> result.opencvTime;
            (*it)["custom_time"] >> result.customTime;
            (*it)["speedup_ratio"] >> result.speedupRatio;
            (*it)["accuracy"] >> result.accuracy;
            (*it)["image_size"] >> result.imageSize;
            (*it)["image_type"] >> result.imageType;
            
            results.push_back(result);
        }
        
        fs.release();
        return results;
    }
};
```
