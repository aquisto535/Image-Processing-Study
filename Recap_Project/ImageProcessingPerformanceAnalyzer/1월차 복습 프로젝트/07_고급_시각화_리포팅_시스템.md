    // 🔄 4분할 결과 비교 뷰어
    void showQuadComparison(const cv::Mat& original, 
                           const cv::Mat& opencv_result,
                           const cv::Mat& custom_result,
                           const cv::Mat& difference_map) {
        
        // 각 이미지를 동일한 크기로 리사이즈
        cv::Size display_size(400, 300);
        cv::Mat orig_resized, opencv_resized, custom_resized, diff_resized;
        
        cv::resize(original, orig_resized, display_size);
        cv::resize(opencv_result, opencv_resized, display_size);
        cv::resize(custom_result, custom_resized, display_size);
        cv::resize(difference_map, diff_resized, display_size);
        
        // 4분할 이미지 생성 (850x650)
        cv::Mat quad_view(cv::Size(850, 650), CV_8UC3, cv::Scalar(255, 255, 255));
        
        // 이미지 배치
        cv::Rect regions[4] = {
            cv::Rect(25, 50, 400, 300),    // 원본 (좌상)
            cv::Rect(425, 50, 400, 300),   // OpenCV 결과 (우상)
            cv::Rect(25, 350, 400, 300),   // Custom 결과 (좌하)
            cv::Rect(425, 350, 400, 300)   // 차이 맵 (우하)
        };
        
        // 이미지 복사
        if (orig_resized.channels() == 1) cv::cvtColor(orig_resized, orig_resized, cv::COLOR_GRAY2BGR);
        if (opencv_resized.channels() == 1) cv::cvtColor(opencv_resized, opencv_resized, cv::COLOR_GRAY2BGR);
        if (custom_resized.channels() == 1) cv::cvtColor(custom_resized, custom_resized, cv::COLOR_GRAY2BGR);
        if (diff_resized.channels() == 1) cv::cvtColor(diff_resized, diff_resized, cv::COLOR_GRAY2BGR);
        
        orig_resized.copyTo(quad_view(regions[0]));
        opencv_resized.copyTo(quad_view(regions[1]));
        custom_resized.copyTo(quad_view(regions[2]));
        diff_resized.copyTo(quad_view(regions[3]));
        
        // 라벨 추가
        cv::putText(quad_view, "Original", cv::Point(200, 40), 
                   cv::FONT_HERSHEY_SIMPLEX, 0.8, cv::Scalar(0, 0, 0), 2);
        cv::putText(quad_view, "OpenCV Result", cv::Point(600, 40), 
                   cv::FONT_HERSHEY_SIMPLEX, 0.8, cv::Scalar(0, 128, 0), 2);
        cv::putText(quad_view, "Custom Result", cv::Point(200, 640), 
                   cv::FONT_HERSHEY_SIMPLEX, 0.8, cv::Scalar(0, 0, 255), 2);
        cv::putText(quad_view, "Difference Map", cv::Point(580, 640), 
                   cv::FONT_HERSHEY_SIMPLEX, 0.8, cv::Scalar(128, 0, 128), 2);
        
        // 경계선 그리기
        for (int i = 0; i < 4; i++) {
            cv::rectangle(quad_view, regions[i], cv::Scalar(128, 128, 128), 2);
        }
        
        cv::imshow("Algorithm Comparison", quad_view);
    }
    
    // 📊 히스토그램 비교 시각화
    void showHistogramComparison(const vector<float>& opencv_hist,
                                const vector<int>& custom_hist) {
        
        cv::Mat hist_chart(cv::Size(800, 600), CV_8UC3, cv::Scalar(255, 255, 255));
        
        // 제목
        cv::putText(hist_chart, "Histogram Comparison", cv::Point(300, 40), 
                   cv::FONT_HERSHEY_SIMPLEX, 1.0, cv::Scalar(0, 0, 0), 2);
        
        // 히스토그램 그리기 영역
        cv::Rect hist_area(50, 80, 700, 400);
        
        // 축 그리기
        cv::line(hist_chart, cv::Point(hist_area.x, hist_area.y + hist_area.height), 
                cv::Point(hist_area.x + hist_area.width, hist_area.y + hist_area.height), 
                cv::Scalar(0, 0, 0), 2);
        cv::line(hist_chart, cv::Point(hist_area.x, hist_area.y), 
                cv::Point(hist_area.x, hist_area.y + hist_area.height), 
                cv::Scalar(0, 0, 0), 2);
        
        // 히스토그램 정규화
        float max_opencv = *max_element(opencv_hist.begin(), opencv_hist.end());
        int max_custom = *max_element(custom_hist.begin(), custom_hist.end());
        float max_val = max(max_opencv, static_cast<float>(max_custom));
        
        // 히스토그램 막대 그리기
        int bin_width = hist_area.width / 256;
        for (int i = 0; i < 256; i++) {
            int x = hist_area.x + i * bin_width;
            
            // OpenCV 히스토그램 (녹색, 투명도 50%)
            int opencv_height = static_cast<int>((opencv_hist[i] / max_val) * hist_area.height);
            cv::rectangle(hist_chart, 
                         cv::Point(x, hist_area.y + hist_area.height - opencv_height),
                         cv::Point(x + bin_width, hist_area.y + hist_area.height),
                         cv::Scalar(0, 255, 0, 128), -1);
            
            // Custom 히스토그램 (빨간색, 라인만)
            int custom_height = static_cast<int>((custom_hist[i] / max_val) * hist_area.height);
            cv::line(hist_chart,
                    cv::Point(x + bin_width/2, hist_area.y + hist_area.height),
                    cv::Point(x + bin_width/2, hist_area.y + hist_area.height - custom_height),
                    cv::Scalar(0, 0, 255), 2);
        }
        
        // 범례
        cv::rectangle(hist_chart, cv::Point(550, 500), cv::Point(570, 520), cv::Scalar(0, 255, 0), -1);
        cv::putText(hist_chart, "OpenCV", cv::Point(580, 515), cv::FONT_HERSHEY_SIMPLEX, 0.6, cv::Scalar(0, 0, 0), 1);
        cv::line(hist_chart, cv::Point(550, 535), cv::Point(570, 535), cv::Scalar(0, 0, 255), 3);
        cv::putText(hist_chart, "Custom", cv::Point(580, 540), cv::FONT_HERSHEY_SIMPLEX, 0.6, cv::Scalar(0, 0, 0), 1);
        
        cv::imshow("Histogram Comparison", hist_chart);
    }
    
    // 📈 성능 트렌드 분석
    void showPerformanceTrends(const PerformanceProfile& profile) {
        cv::Mat trend_chart(cv::Size(1000, 700), CV_8UC3, cv::Scalar(255, 255, 255));
        
        drawTitle(trend_chart, "Performance Trends Over Time", cv::Point(500, 40));
        
        // 트렌드 차트 영역
        cv::Rect chart_area(80, 100, 840, 500);
        
        // 축 그리기
        drawAxes(trend_chart, chart_area, "Test Iterations", "Performance Ratio");
        
        // 각 알고리즘별 트렌드 라인 그리기
        int color_index = 0;
        for (const auto& [algorithm, times] : profile.algorithm_times) {
            if (times.size() < 2) continue;
            
            vector<cv::Point2f> trend_points;
            
            // 데이터 포인트 계산
            for (int i = 0; i < times.size(); i++) {
                float x = chart_area.x + (float)i / (times.size() - 1) * chart_area.width;
                float y = chart_area.y + chart_area.height - (times[i] / 50.0) * chart_area.height; // 50ms 최대값 가정
                trend_points.push_back(cv::Point2f(x, y));
            }
            
            // 트렌드 라인 그리기
            cv::Scalar color = color_palette[color_index % color_palette.size()];
            for (int i = 1; i < trend_points.size(); i++) {
                cv::line(trend_chart, trend_points[i-1], trend_points[i], color, 2);
            }
            
            // 데이터 포인트 마커
            for (const auto& point : trend_points) {
                cv::circle(trend_chart, point, 4, color, -1);
            }
            
            color_index++;
        }
        
        // 범례 그리기
        drawTrendLegend(trend_chart, cv::Point(750, 150), profile.algorithm_times);
        
        cv::imshow("Performance Trends", trend_chart);
    }
    
private:
    // 🎨 제목 그리기
    void drawTitle(cv::Mat& image, const string& title, cv::Point center) {
        int baseline = 0;
        cv::Size text_size = cv::getTextSize(title, cv::FONT_HERSHEY_SIMPLEX, 1.2, 2, &baseline);
        
        cv::Point text_origin(center.x - text_size.width/2, center.y + text_size.height/2);
        cv::putText(image, title, text_origin, cv::FONT_HERSHEY_SIMPLEX, 1.2, cv::Scalar(0, 0, 0), 2);
        
        // 제목 밑줄
        cv::line(image, 
                cv::Point(text_origin.x, text_origin.y + 10),
                cv::Point(text_origin.x + text_size.width, text_origin.y + 10),
                cv::Scalar(100, 100, 100), 2);
    }
    
    // 📊 축 그리기
    void drawAxes(cv::Mat& image, cv::Rect area, const string& x_label, const string& y_label) {
        // X축
        cv::line(image, cv::Point(area.x, area.y + area.height), 
                cv::Point(area.x + area.width, area.y + area.height), 
                cv::Scalar(0, 0, 0), 2);
        
        // Y축
        cv::line(image, cv::Point(area.x, area.y), 
                cv::Point(area.x, area.y + area.height), 
                cv::Scalar(0, 0, 0), 2);
        
        // 격자선 그리기
        drawGrid(image, area);
        
        // 축 라벨
        cv::putText(image, x_label, 
                   cv::Point(area.x + area.width/2 - 50, area.y + area.height + 40), 
                   cv::FONT_HERSHEY_SIMPLEX, 0.7, cv::Scalar(0, 0, 0), 1);
        
        // Y축 라벨 (90도 회전 효과)
        cv::putText(image, y_label, 
                   cv::Point(area.x - 70, area.y + area.height/2), 
                   cv::FONT_HERSHEY_SIMPLEX, 0.7, cv::Scalar(0, 0, 0), 1);
    }
    
    // 🔳 격자선 그리기
    void drawGrid(cv::Mat& image, cv::Rect area) {
        // 수직 격자선
        for (int i = 1; i < 10; i++) {
            int x = area.x + i * area.width / 10;
            cv::line(image, cv::Point(x, area.y), cv::Point(x, area.y + area.height), 
                    settings.grid_color, 1);
        }
        
        // 수평 격자선
        for (int i = 1; i < 10; i++) {
            int y = area.y + i * area.height / 10;
            cv::line(image, cv::Point(area.x, y), cv::Point(area.x + area.width, y), 
                    settings.grid_color, 1);
        }
    }
    
    // 📊 성능 막대 그래프 그리기
    void drawPerformanceBars(cv::Mat& image, cv::Rect area, const vector<BenchmarkResult>& results) {
        if (results.empty()) return;
        
        int bar_width = area.width / (results.size() * 2 + 1);
        double max_time = 0;
        
        // 최대값 찾기
        for (const auto& result : results) {
            max_time = max(max_time, max(result.opencvTime, result.customTime));
        }
        
        for (int i = 0; i < results.size(); i++) {
            int x_base = area.x + (i * 2 + 1) * bar_width;
            
            // OpenCV 막대 (녹색)
            int opencv_height = static_cast<int>((results[i].opencvTime / max_time) * area.height);
            cv::rectangle(image, 
                         cv::Point(x_base, area.y + area.height - opencv_height),
                         cv::Point(x_base + bar_width - 5, area.y + area.height),
                         cv::Scalar(0, 200, 0), -1);
            
            // Custom 막대 (빨간색)
            int custom_height = static_cast<int>((results[i].customTime / max_time) * area.height);
            cv::rectangle(image, 
                         cv::Point(x_base + bar_width, area.y + area.height - custom_height),
                         cv::Point(x_base + 2 * bar_width - 5, area.y + area.height),
                         cv::Scalar(0, 0, 200), -1);
            
            // 알고리즘 이름 (회전)
            cv::putText(image, results[i].algorithmName.substr(0, 8), 
                       cv::Point(x_base, area.y + area.height + 20), 
                       cv::FONT_HERSHEY_SIMPLEX, 0.5, cv::Scalar(0, 0, 0), 1);
        }
    }
};
